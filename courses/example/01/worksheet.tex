\problem{次の値を求めなさい。\quad $2^3 + 3^2$}
\answerbox[2cm]{$8 + 9 = 17$}

\problem{関数 $f(x) = x^2$ の $x = 3$ における微分係数を求めなさい。}
\answerbox[3cm]{$f'(x) = 2x$ より $f'(3) = 6$}
