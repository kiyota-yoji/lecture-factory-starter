    機械学習の利点として正しいものはどれですか？
    \begin{choices}
      \wrongchoice{必ず正しい答えを出せる}
      \wrongchoice{すべてのデータに対応できる}
      \correctchoice{データからパターンを見つけ出せる}
      \wrongchoice{人間の知識が不要になる}
    \end{choices}
  \end{question}
}

\element{aib10}{
  \begin{question}{AIB10-Q2}
    統計的機械学習が使われる場面として、もっとも適切なものはどれですか？
    \begin{choices}
      \wrongchoice{人間の感情を直接読み取る}
      \correctchoice{データにもとづいて予測を行う}
      \wrongchoice{仮説を証明するためのルールを作る}
      \wrongchoice{物理的な実験を完全に置きかえる}
    \end{choices}
  \end{question}
}

\element{aib10}{
  \begin{question}{AIB10-Q3}
    強化学習の特徴として正しいものはどれですか？
    \begin{choices}
      \wrongchoice{ラベル付きのデータを用いる}
      \correctchoice{環境とのやりとりを通じて学習する}
      \wrongchoice{データの次元を削減する}
      \wrongchoice{すべての行動が正解である}
    \end{choices}
  \end{question}
}

\element{aib10}{
  \begin{question}{AIB10-Q4}
    機械学習の性質として正しいものはどれですか？
    \begin{choices}
      \wrongchoice{すべてのモデルは、判断の理由を説明できる}
      \correctchoice{学習に使うデータに偏りがあると、不公平な判断が反映される}
      \wrongchoice{機械学習には大量のデータは必要ない}
      \wrongchoice{すべての問題に使うことができる}
    \end{choices}
  \end{question}
}

\element{aib10}{
  \begin{question}{AIB10-Q5}
    ChatGPT のような大規模言語モデル（生成 AI）についての説明として、もっとも適切なものはどれですか？
    \begin{choices}
      \wrongchoice{人がルールを1つずつ書き込んで作られているので、ルールにない質問には答えられない}
      \correctchoice{学んだ大量の文章をもとに自然な文章を作れるが、学んだ時点より後の出来事や前例のない問題は苦手なので、調べたり確かめたりする必要がある}
      \wrongchoice{人間のフィードバックで調整されているので、まちがった答えを出すことはない}
      \wrongchoice{正解ラベルのない文章からは何も学べないので、学習に使う文章にはすべて人が正解を付けている}
    \end{choices}
  \end{question}
}
