\element{bm11}{
  \begin{question}{BM11-Q1}
    $\begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix} + \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}$ は？
    \begin{choiceshoriz}
      \correctchoice{$\begin{pmatrix} 2 & 2 \\ 3 & 5 \end{pmatrix}$}
      \wrongchoice{$\begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$}
      \wrongchoice{$\begin{pmatrix} 2 & 3 \\ 4 & 5 \end{pmatrix}$}
      \wrongchoice{$\begin{pmatrix} 1 & 0 \\ 0 & 4 \end{pmatrix}$}
    \end{choiceshoriz}
  \end{question}
}

\element{bm11}{
  \begin{question}{BM11-Q2}
    $\begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}\begin{pmatrix} 1 \\ 1 \end{pmatrix}$ は？
    \begin{choiceshoriz}
      \correctchoice{$\begin{pmatrix} 3 \\ 7 \end{pmatrix}$}
      \wrongchoice{$\begin{pmatrix} 4 \\ 6 \end{pmatrix}$}
      \wrongchoice{$\begin{pmatrix} 1 \\ 4 \end{pmatrix}$}
      \wrongchoice{$\begin{pmatrix} 2 \\ 5 \end{pmatrix}$}
    \end{choiceshoriz}
  \end{question}
}

\element{bm11}{

      \wrongchoice{$\begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}$}
