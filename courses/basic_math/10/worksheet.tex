\problem{$\vec{a} = (2,\, 1)$，$\vec{b} = (-1,\, 3)$ のとき、次を求めなさい。}
\begin{minipage}[t]{0.23\linewidth}
  (1) $\vec{a} + \vec{b}$
  \answerbox[2.2cm]{$(1,\, 4)$}
\end{minipage}\hfill
\begin{minipage}[t]{0.23\linewidth}
  (2) $3\vec{a}$
  \answerbox[2.2cm]{$(6,\, 3)$}
\end{minipage}\hfill
\begin{minipage}[t]{0.23\linewidth}
  (3) $2\vec{a} - \vec{b}$
  \answerbox[2.2cm]{$(5,\, -1)$}
\end{minipage}\hfill
\begin{minipage}[t]{0.23\linewidth}
  (4) $|\vec{b}|$
  \answerbox[2.2cm]{$\sqrt{1 + 9} = \sqrt{10}$}
\end{minipage}

\problem{次の式を、1つのベクトルで表しなさい。}
\begin{minipage}[t]{0.48\linewidth}
  (1) $\overrightarrow{\mathrm{AB}} + \overrightarrow{\mathrm{BC}} + \overrightarrow{\mathrm{CD}}$
  \answerbox[2.2cm]{$\overrightarrow{\mathrm{AD}}$}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}
  (2) $\overrightarrow{\mathrm{AB}} - \overrightarrow{\mathrm{CB}}$
  \answerbox[2.2cm]{$\overrightarrow{\mathrm{AB}} + \overrightarrow{\mathrm{BC}} = \overrightarrow{\mathrm{AC}}$}
\end{minipage}

\problem{2点 A$(1,\, 2)$，B$(7,\, 5)$ について、次を求めなさい。}
\begin{minipage}[t]{0.31\linewidth}
  (1) $\overrightarrow{\mathrm{AB}}$ の成分
  \answerbox[3cm]{$(7 - 1,\ 5 - 2) = (6,\, 3)$}
\end{minipage}\hfill
\begin{minipage}[t]{0.31\linewidth}
  (2) AB を $2:1$ に内分する点
  \answerbox[3cm]{$\dfrac{1 \cdot (1,\,2) + 2 \cdot (7,\,5)}{3}$\\[0.5ex] $= (5,\, 4)$}
\end{minipage}\hfill
\begin{minipage}[t]{0.31\linewidth}
  (3) 線分 AB の中点
  \answerbox[3cm]{$\left(4,\ \dfrac{7}{2}\right)$}
\end{minipage}

\problem{3点 A$(0,\, 0)$，B$(6,\, 0)$，C$(3,\, 6)$ を頂点とする三角形の重心の座標を求めなさい。}
\answerbox[2.5cm]{$\left(\dfrac{0 + 6 + 3}{3},\ \dfrac{0 + 0 + 6}{3}\right) = (3,\, 2)$}

\problem{物体に 2つの力 $\vec{F}_1 = (2,\, 3)$，$\vec{F}_2 = (4,\, -1)$（単位は N）がはたらいている。合力とその大きさを求めなさい。}
\answerbox[3cm]{合力 $\vec{F}_1 + \vec{F}_2 = (6,\, 2)$，\ 大きさ $\sqrt{36 + 4} = \sqrt{40} = 2\sqrt{10}$ N}
